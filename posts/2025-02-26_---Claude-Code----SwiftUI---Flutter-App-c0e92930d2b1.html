以下彼得潘以開發 SwiftUI / Flutter 猜數字 App 為例，一步步說明 Claude Code 的相關操作。

npm install -g @anthropic-ai/claude-code

sudo npm install -g @anthropic-ai/claude-code

以下我們請 Claude 寫一個可愛的猜數字 App。

Claude Code 直接修改資料夾下的 ContentView.swift，綠色是新增的程式，紅色是刪除的程式。

選擇 Yes 同意 Claude 的修改。

回到 Xcode，Claude Code 成功生成猜數字 App。

完整程式如下。

import SwiftUI

struct ContentView: View {
    @State private var targetNumber = Int.random(in: 1...100)
    @State private var userGuess = ""
    @State private var message = "猜一個 1 到 100 之間的數字！"
    @State private var emoji = "🎮"
    @State private var attemptCount = 0
    @State private var gameOver = false
    @State private var showConfetti = false
    
    var body: some View {
        ZStack {
            Color(red: 0.95, green: 0.95, blue: 1.0)
                .ignoresSafeArea()
            
            VStack(spacing: 20) {
                Text("🎲 猜數字遊戲 🎲")
                    .font(.largeTitle)
                    .fontWeight(.bold)
                    .foregroundColor(.purple)
                
                Text(emoji)
                    .font(.system(size: 80))
                
                Text(message)
                    .font(.title2)
                    .foregroundColor(.indigo)
                    .multilineTextAlignment(.center)
                    .padding()
                    .frame(height: 80)
                
                if !gameOver {
                    TextField("請輸入數字", text: $userGuess)
                        .keyboardType(.numberPad)
                        .padding()
                        .background(Color.white)
                        .cornerRadius(10)
                        .shadow(radius: 2)
                        .padding(.horizontal)
                    
                    Button(action: checkGuess) {
                        Text("猜一猜")
                            .font(.title2)
                            .padding()
                            .frame(maxWidth: .infinity)
                            .background(Color.purple)
                            .foregroundColor(.white)
                            .cornerRadius(10)
                            .shadow(radius: 2)
                    }
                    .padding(.horizontal)
                }
                
                Text("已嘗試次數: \(attemptCount)")
                    .foregroundColor(.gray)
                    .padding(.top)
                
                if gameOver {
                    Button(action: resetGame) {
                        Text("再玩一次")
                            .font(.title2)
                            .padding()
                            .frame(maxWidth: .infinity)
                            .background(Color.green)
                            .foregroundColor(.white)
                            .cornerRadius(10)
                            .shadow(radius: 2)
                    }
                    .padding(.horizontal)
                    .padding(.top)
                }
            }
            .padding()
        }
    }
    
    func checkGuess() {
        guard let guess = Int(userGuess) else {
            message = "請輸入有效的數字！"
            emoji = "❓"
            return
        }
        
        attemptCount += 1
        
        if guess < 1 || guess > 100 {
            message = "請輸入 1 到 100 之間的數字！"
            emoji = "🧐"
        } else if guess < targetNumber {
            message = "太小了！再大一點"
            emoji = "👆"
        } else if guess > targetNumber {
            message = "太大了！再小一點"
            emoji = "👇"
        } else {
            message = "恭喜你猜對了！\n答案就是 \(targetNumber)"
            emoji = "🎉"
            gameOver = true
            showConfetti = true
        }
        
        userGuess = ""
    }
    
    func resetGame() {
        targetNumber = Int.random(in: 1...100)
        message = "猜一個 1 到 100 之間的數字！"
        emoji = "🎮"
        attemptCount = 0
        userGuess = ""
        gameOver = false
        showConfetti = false
    }
}

  • 開發 Flutter 猜數字 App。

以下是另一個例子，Claude Code 修改 main.dart 實現 Flutter 版的猜數字 App。

完整程式如下。

import 'package:flutter/material.dart';
import 'dart:math';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: '猜數字遊戲',
      theme: ThemeData(
        colorScheme: ColorScheme.fromSeed(seedColor: Colors.pink),
        useMaterial3: true,
        fontFamily: 'Roboto',
      ),
      home: const NumberGuessGame(title: '可愛的猜數字遊戲'),
    );
  }
}

class NumberGuessGame extends StatefulWidget {
  const NumberGuessGame({super.key, required this.title});

  final String title;

  @override
  State<NumberGuessGame> createState() => _NumberGuessGameState();
}

class _NumberGuessGameState extends State<NumberGuessGame> {
  // 遊戲數據
  final int _maxNumber = 100;
  late int _targetNumber;
  late int _minGuess;
  late int _maxGuess;
  final TextEditingController _guessController = TextEditingController();
  String _feedbackMessage = '請猜一個數字吧！';
  bool _hasWon = false;
  int _attemptCount = 0;
  
  // 表情列表
  final List<String> _happyEmojis = ['(◕‿◕)', '(ﾉ◕ヮ◕)ﾉ*:･ﾟ✧', '(づ｡◕‿‿◕｡)づ', '(✿◠‿◠)', 'ʕ•ᴥ•ʔ'];
  final List<String> _sadEmojis = ['(◕︵◕)', '(っ˘̩╭╮˘̩)っ', '(｡•́︿•̀｡)', '(╥﹏╥)', '(◞‸◟)'];
  final List<String> _excitedEmojis = ['(≧◡≦)', '(ﾉ≧∀≦)ﾉ', '(*≧ω≦*)', '(〜￣▽￣)〜', '(ﾉ´ヮ`)ﾉ*: ･ﾟ'];
  
  @override
  void initState() {
    super.initState();
    _startNewGame();
  }
  
  void _startNewGame() {
    setState(() {
      // 產生 1 到 _maxNumber 之間的隨機數
      _targetNumber = Random().nextInt(_maxNumber) + 1;
      _minGuess = 1;
      _maxGuess = _maxNumber;
      _feedbackMessage = '請猜一個 1 到 $_maxNumber 之間的數字吧！';
      _hasWon = false;
      _attemptCount = 0;
      _guessController.clear();
    });
  }
  
  void _makeGuess() {
    // 檢查輸入是否為數字
    if (_guessController.text.isEmpty) {
      setState(() {
        _feedbackMessage = '請輸入一個數字！ ${_sadEmojis[Random().nextInt(_sadEmojis.length)]}';
      });
      return;
    }
    
    final guess = int.tryParse(_guessController.text);
    if (guess == null) {
      setState(() {
        _feedbackMessage = '請輸入一個有效的數字！ ${_sadEmojis[Random().nextInt(_sadEmojis.length)]}';
      });
      return;
    }
    
    setState(() {
      _attemptCount++;
      
      if (guess < _minGuess || guess > _maxGuess) {
        _feedbackMessage = '請在範圍內猜測：$_minGuess 到 $_maxGuess 之間 ${_sadEmojis[Random().nextInt(_sadEmojis.length)]}';
      } else if (guess < _targetNumber) {
        _minGuess = guess + 1;
        _feedbackMessage = '太小了！再猜大一點，在 $_minGuess 到 $_maxGuess 之間 ${_happyEmojis[Random().nextInt(_happyEmojis.length)]}';
      } else if (guess > _targetNumber) {
        _maxGuess = guess - 1;
        _feedbackMessage = '太大了！再猜小一點，在 $_minGuess 到 $_maxGuess 之間 ${_happyEmojis[Random().nextInt(_happyEmojis.length)]}';
      } else {
        _hasWon = true;
        _feedbackMessage = '恭喜你猜對了！答案就是 $_targetNumber！\n你總共猜了 $_attemptCount 次 ${_excitedEmojis[Random().nextInt(_excitedEmojis.length)]}';
      }
      
      _guessController.clear();
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        backgroundColor: Theme.of(context).colorScheme.inversePrimary,
        title: Text(widget.title),
      ),
      body: Container(
        decoration: BoxDecoration(
          gradient: LinearGradient(
            begin: Alignment.topCenter,
            end: Alignment.bottomCenter,
            colors: [Colors.pink.shade50, Colors.purple.shade50],
          ),
        ),
        child: Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(20.0),
            child: Column(
              mainAxisAlignment: MainAxisAlignment.center,
              children: <Widget>[
                // 可愛的圖像
                Icon(
                  _hasWon ? Icons.celebration : Icons.psychology,
                  size: 100,
                  color: _hasWon ? Colors.amber : Colors.pinkAccent,
                ),
                const SizedBox(height: 20),
                
                // 遊戲訊息
                Container(
                  padding: const EdgeInsets.all(20.0),
                  decoration: BoxDecoration(
                    color: Colors.white,
                    borderRadius: BorderRadius.circular(20),
                    boxShadow: [
                      BoxShadow(
                        color: Colors.grey.withOpacity(0.3),
                        spreadRadius: 2,
                        blurRadius: 5,
                        offset: const Offset(0, 3),
                      ),
                    ],
                  ),
                  child: Text(
                    _feedbackMessage,
                    textAlign: TextAlign.center,
                    style: TextStyle(
                      fontSize: 20,
                      fontWeight: FontWeight.bold,
                      color: _hasWon ? Colors.green : Colors.deepPurple,
                    ),
                  ),
                ),
                const SizedBox(height: 30),
                
                // 猜測輸入區
                if (!_hasWon) ...[
                  Container(
                    width: 200,
                    decoration: BoxDecoration(
                      color: Colors.white,
                      borderRadius: BorderRadius.circular(15),
                      boxShadow: [
                        BoxShadow(
                          color: Colors.grey.withOpacity(0.2),
                          spreadRadius: 1,
                          blurRadius: 3,
                          offset: const Offset(0, 2),
                        ),
                      ],
                    ),
                    child: TextField(
                      controller: _guessController,
                      keyboardType: TextInputType.number,
                      textAlign: TextAlign.center,
                      style: const TextStyle(fontSize: 24),
                      decoration: InputDecoration(
                        hintText: '輸入數字',
                        hintStyle: TextStyle(color: Colors.grey.shade400),
                        border: OutlineInputBorder(
                          borderRadius: BorderRadius.circular(15),
                          borderSide: BorderSide.none,
                        ),
                      ),
                      onSubmitted: (_) => _makeGuess(),
                    ),
                  ),
                  const SizedBox(height: 20),
                  
                  // 猜測按鈕
                  ElevatedButton(
                    onPressed: _makeGuess,
                    style: ElevatedButton.styleFrom(
                      backgroundColor: Colors.pinkAccent,
                      foregroundColor: Colors.white,
                      padding: const EdgeInsets.symmetric(horizontal: 40, vertical: 12),
                      shape: RoundedRectangleBorder(
                        borderRadius: BorderRadius.circular(30),
                      ),
                    ),
                    child: const Text(
                      '猜！',
                      style: TextStyle(fontSize: 20),
                    ),
                  ),
                ] else ...[
                  // 再玩一次按鈕
                  ElevatedButton(
                    onPressed: _startNewGame,
                    style: ElevatedButton.styleFrom(
                      backgroundColor: Colors.green,
                      foregroundColor: Colors.white,
                      padding: const EdgeInsets.symmetric(horizontal: 40, vertical: 12),
                      shape: RoundedRectangleBorder(
                        borderRadius: BorderRadius.circular(30),
                      ),
                    ),
                    child: const Text(
                      '再玩一次！',
                      style: TextStyle(fontSize: 20),
                    ),
                  ),
                ],
              ],
            ),
          ),
        ),
      ),
    );
  }
  
  @override
  void dispose() {
    _guessController.dispose();
    super.dispose();
  }
}
