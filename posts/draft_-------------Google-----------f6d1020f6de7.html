我想要每週生成內容一樣的 Google 表單，有什麼好方法



https://docs.google.com/forms/d/1wNiOiCwwhB42EEyMTCdLUJYJdrMFGAJjF0qPR21211o/edit






寫一個 Google Apps Script，功能如下。
1.複製某個 Google 表單
https://docs.google.com/forms/d/1wNiOiCwwhB42EEyMTCdLUJYJdrMFGAJjF0qPR21211o/edit
2.表單的名稱和標題格式為"App 課程問卷 日期"，日期是複製表單的日期，比方"App 課程問卷 2026/6/8"
3.複製表單後，自動建立新的試算表並連結
4.將表單發布並設成公開分享
5.將試算表設成公開分享
6.將表單跟試算表的分享連結寄到 apppeterpan@gmail.com, 信件標題寫表單名字



/**
 * 複製 Google 表單、建立連結試算表、設公開分享並寄信通知
 * 
 * 使用方式：
 *   在 Google Apps Script 編輯器中貼上此程式碼，執行 copyFormAndSetup()
 * 
 * 所需權限（第一次執行時會要求授權）：
 *   - Google Forms (DriveApp / FormApp)
 *   - Google Drive
 *   - Gmail
 */

// ─────────────────────────────────────────────
// 設定區
// ─────────────────────────────────────────────
const SOURCE_FORM_ID    = "1wNiOiCwwhB42EEyMTCdLUJYJdrMFGAJjF0qPR21211o";
const NOTIFY_EMAIL      = "apppeterpan@gmail.com";
const FORM_TITLE_PREFIX = "App 課程問卷";


// ─────────────────────────────────────────────
// 主程式
// ─────────────────────────────────────────────
function copyFormAndSetup() {

  // ① 產生今日日期字串，格式 YYYY/M/D（無補零）
  const today     = new Date();
  const dateStr   = `${today.getFullYear()}/${today.getMonth() + 1}/${today.getDate()}`;
  const formTitle = `${FORM_TITLE_PREFIX} ${dateStr}`;

  // ② 複製來源表單
  const sourceFile = DriveApp.getFileById(SOURCE_FORM_ID);
  const newFile    = sourceFile.makeCopy(formTitle);
  const newFormId  = newFile.getId();
  const newForm    = FormApp.openById(newFormId);

  // ③ 更新表單標題（顯示在表單頂端）與文件名稱
  newForm.setTitle(formTitle);
  newFile.setName(formTitle);

  // ④ 建立新試算表並與表單連結（Linked Sheet）
  const spreadsheet = SpreadsheetApp.create(formTitle);
  newForm.setDestination(FormApp.DestinationType.SPREADSHEET, spreadsheet.getId());

  // ⑤ 發布表單
  newForm.setPublished(true);
  newForm.setAcceptingResponses(true);

  // ⑥ 將表單設為「知道連結的人可檢視 / 填寫」
  const formFile = DriveApp.getFileById(newForm.getId());
  formFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

  // ⑦ 取得表單的公開填寫連結
  const formPublishedUrl = newForm.getPublishedUrl();

  // ⑧ 將試算表設為「任何人都可以用連結檢視」
  const ssFile = DriveApp.getFileById(spreadsheet.getId());
  ssFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

  // ⑨ 取得試算表的分享連結
  const ssUrl = `https://docs.google.com/spreadsheets/d/${spreadsheet.getId()}/edit?usp=sharing`;

  // ⑩ 寄送通知信
  const subject = formTitle;
  const body    = `表單連結：${formPublishedUrl}\n試算表連結：${ssUrl}`;
  GmailApp.sendEmail(NOTIFY_EMAIL, subject, body);

  // ⑪ 在執行記錄顯示結果
  Logger.log("✅ 完成！");
  Logger.log(`表單名稱：${formTitle}`);
  Logger.log(`表單填寫連結：${formPublishedUrl}`);
  Logger.log(`試算表連結：${ssUrl}`);
  Logger.log(`通知信已寄至：${NOTIFY_EMAIL}`);
}
