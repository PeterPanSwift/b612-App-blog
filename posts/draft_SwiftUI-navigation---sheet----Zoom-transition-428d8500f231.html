SwiftUI navigation & sheet 搭配 Zoom transition


navigation

.navigationTransition(.zoom(sourceID: landmark.name, in: namespace))

struct ContentView: View {
    @Namespace private var namespace
    let landmarks = [
        Landmark(name: "天空之湖溫泉", intro: "天空之湖溫泉（Sky Lagoon）位於冰島雷克雅維克近郊，是一座面向北大西洋的地熱溫泉。無邊際的溫泉池與海天一線的壯闊景色融為一體，讓人彷彿浸泡在天空之中。結合冰島傳統的七步沐浴儀式，帶來身心靈的極致放鬆體驗。"),
        Landmark(name: "鑽石沙灘", intro: "鑽石沙灘（Diamond Beach）位於冰島東南部傑古沙龍冰河湖（Jökulsárlón）旁，黑色火山砂灘上散落著大大小小晶瑩剔透的冰塊，在陽光照射下閃爍如鑽石般的光芒，因此得名。這裡是攝影愛好者的天堂，冰與火交織的景象令人嘆為觀止。"),
    ]
    
    var body: some View {
        NavigationStack {
            List {
                ForEach(landmarks) { landmark in
                    NavigationLink {
                        VStack {
                            Image(landmark.name)
                                .resizable()
                                .scaledToFit()
                            Text(landmark.intro)
                        }
                        .navigationTransition(.zoom(sourceID: landmark.name, in: namespace))
                       
                    } label: {
                        HStack {
                            Image(landmark.name)
                                .resizable()
                                .scaledToFit()
                                .frame(width: 100, height: 100)
                            Text(landmark.name)
                                .font(.largeTitle)
                        }
                    }

                }
            }
        }
       
       
    }
}

sheet

struct ContentView: View {
    @Namespace private var namespace
    @State private var isPresentingAddLandmark = false
    
    var body: some View {
        NavigationStack {
            List {
            }
            .toolbar {
                Button("Add") {
                    isPresentingAddLandmark = true
                }
                .matchedTransitionSource(id: "addLandmark", in: namespace)
            }
            .sheet(isPresented: $isPresentingAddLandmark) {
                Image(systemName: "star")
                    .resizable()
                    .scaledToFit()
                    .navigationTransition(.zoom(sourceID: "addLandmark", in: namespace))
            }
        }
       
       
    }
}
