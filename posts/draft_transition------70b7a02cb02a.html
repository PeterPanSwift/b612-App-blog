transition 翻牌動畫


struct ContentView: View {
    @State private var isEarned = false
   
    var body: some View {
        Group {
            if isEarned {
                Image(.鑽石沙灘)
                    .resizable()
                    
            } else {
                Image(.天空之湖溫泉)
                    .resizable()
            }
        }
        .scaledToFill()
        .frame(width: 300, height: 300)
        .clipped()
        .transition(CustomFlipTransition())
        .onTapGesture {
            withAnimation {
                isEarned.toggle()
            }
        }
    }
}



struct CustomFlipTransition: Transition {
    func body(content: Content, phase: TransitionPhase) -> some View {
        content
            .rotation3DEffect(
                .degrees(phase.isIdentity ? 0 : (phase.value * 180)),
                axis: (x: 0, y: 1, z: 0)
            )
            .opacity(phase.isIdentity ? 1 : 0)
    }
}
