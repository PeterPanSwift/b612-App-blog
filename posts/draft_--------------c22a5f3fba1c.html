    @IBAction func redoBtn(_ sender: Any) {
        questionIndex = 0
        score = 0
        questionArray.shuffle()
        putAnswer(number: 0)
        syncScoreAndLabel(scoreText: "0", numberOfQuestion: 1)
        hideOptions(hide: false)
        nextBtnOutlet.isHidden = false
    }
    @IBAction func nextBtn(_ sender: Any) {
        questionIndex = (questionIndex + 1) % questionArray.count
        // 重返回第一題
        if questionIndex == 0 {
            questionLabel.text = "恭喜你獲得\(score)分!"
            hideOptions(hide: true)
            nextBtnOutlet.isHidden = true
        } else{
            putAnswer(number: questionIndex)
            rightAnswer = questionArray[questionIndex].answer
            numberOfQ.text = String(format: "%02d" + "/10", questionIndex+1)
            hideOptions(hide: false)
        }
    }
    //填入題目與答案選項
    func putAnswer(number: Int) {
        questionLabel.text = questionArray[number].description
        for i in 0...2 {
            //答案選項丟進按鈕title
            answerButtons[i].setTitle(questionArray[number].options[i], for: .normal)
        }
        rightAnswer = questionArray[number].answer
    }
    
    @IBAction func correctAnswer(_ sender: UIButton) {
        questionIndex = (questionIndex + 1) % questionArray.count
        let userAnswer = sender.currentTitle!
        //第十題按下按鈕時
        if questionIndex == 0{
            //儲存最後一題分數
            if userAnswer == rightAnswer {
                score += 10
            } else if score > 0{
                score -= 5
            }
            questionLabel.text = "恭喜你獲得\(score)分!"
            scoreLabel.text = String(score)
            nextBtnOutlet.isHidden = true
            hideOptions(hide: true)
        } else {
            if rightAnswer != userAnswer {
                if score > 0 {
                    score -= 5
                } else {
                    score = 0
                }
            } else {
                score += 10
            }
            syncScoreAndLabel(scoreText: String(score), numberOfQuestion: questionIndex+1)
            putAnswer(number: questionIndex)
        }
    }
    
    func syncScoreAndLabel(scoreText: String, numberOfQuestion: Int){
            scoreLabel.text = scoreText
            numberOfQ.text = String(format: "%02d" + "/10", numberOfQuestion)
    }
    func hideOptions(hide: Bool){
            for i in 0...2 {
                answerButtons[i].isHidden = hide
            }
    }
}

此程式是一個簡單的 iOS App 遊戲，題目為科幻電影相關問題，使用者需在三個選項中選出正確答案，並且在遊戲結束後顯示分數。

以下為程式概述：

在程式中 import UIKit，將遊戲畫面的元件放置在 ViewController 這個類別中。
建立了一個 Question 的結構體，儲存題目的描述、正確答案以及選項。
在 ViewController 中定義了各個元件的 IBOutlet 變數，並建立了一個 questionArray 陣列，用來儲存所有的題目。
在 viewDidLoad 函數中，建立了 10 個題目並將它們添加到 questionArray 陣列中，並使用 shuffle 方法將題目隨機排序。接著呼叫 putAnswer 函數，將第一個題目的問題描述和選項放入畫面中，並將 rightAnswer 設為第一個題目的正確答案。
在 redoBtn 函數中，重新開始遊戲，將 questionIndex 和 score 設為 0，將 questionArray 陣列重新排序，呼叫 putAnswer 函數，並將分數歸零，將畫面上的選項隱藏並顯示下一題按鈕。
在 nextBtn 函數中，每當使用者點擊下一題按鈕，將 questionIndex 加一。如果 questionIndex 等於 questionArray 的元素個數，代表所有題目已經回答完畢，此時顯示分數並隱藏選項和下一題按鈕。否則，呼叫 putAnswer 函數，將下一個題目的問題描述和選項放入畫面中，並將 rightAnswer 設為下一個題目的正確答案。
putAnswer 函數接受一個數字作為引數，代表要顯示第幾個題目。此函數會將題目的問題描述和選項放入畫面中。
syncScoreAndLabel 函數接受兩個引數，代表要顯示的分數和題號，此函數會將分數和題號放入畫面中。
hideOptions 函數接受一個布林值作為引數，當引數為 true 時
