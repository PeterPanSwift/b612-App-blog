struct ContentView: View {
    @State private var scale = 1.0
   
    var body: some View {
        VStack {
            Image(systemName: "star")
                .resizable()
                .frame(width: 100, height: 100)
                .scaleEffect(scale)
                .animation(.bouncy(extraBounce: 0.5), value: scale)
            Button("button") {
                scale += 1
            }
        }
    }
}
