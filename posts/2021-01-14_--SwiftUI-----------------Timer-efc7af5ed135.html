方法 2: 使用 Timer

  • 建立 SwiftUI App 專案，將 ContentView 原本的內容清除，貼上以下程式。
import SwiftUI

extension String {
    subscript(_ i: Int) -> String {
        let idx1 = index(startIndex, offsetBy: i)
        let idx2 = index(idx1, offsetBy: 1)
        return String(self[idx1..<idx2])
    }
}

struct ContentView: View {
    @State private var content = ""
    @State private var showContent = ""
    @State private var iArray = [Int]()
    @State private var jArray = [Int]()
    let timer = Timer.publish(every: 2, on: .main, in: .common).autoconnect()
    
    func draw() {
       
    }
    
    var body: some View {
        VStack(spacing: 20) {
            HStack(spacing: 20) {
                if showContent.count > 0 {
                    let index = showContent.count - 1
                    Text("i \(iArray[index])")
                    Text("j \(jArray[index])")
                    Text("加入 \(showContent[index] == "\n" ? "換行" : showContent[index])")
                }
               
            }
            Text(showContent)
            Spacer()
        }
        .font(.largeTitle)
        .onAppear(perform: {
            draw()
        })
        .onReceive(timer, perform: { (_) in
            if content.count > 0 {
                showContent += String(content.removeFirst())
            } else {
                timer.upstream.connect().cancel()
            }
        })
    }
}

struct ContentView_Previews: PreviewProvider {
    static var previews: some View {
        ContentView()
    }
}
  • 在 function draw 裡輸入畫圖的程式。
func draw() {
        let count = 3
        for i in 1...count {
            for j in 1...count {
                if i > j {
                    content = content + "🐟"
                } else {
                    content = content + "🐠"
                }
                iArray.append(i)
                jArray.append(j)
            }
            content = content + "\n"
            iArray.append(iArray.last!)
            jArray.append(jArray.last!)
        }
}

值得注意的，為了在畫面上顯示 i & j 的數字，請在 content 增加字串時，加入以下兩行程式。

iArray.append(i)
jArray.append(j)
  • 執行 App。

輸入程式後，右邊的 preview 畫面將自動更新執行。

若想調整迴圈執行的速度，可修改以下參數 every 的數字，比方 2 代表每 2 秒鐘新增添加的字串。

let timer = Timer.publish(every: 2, on: .main, in: .common).autoconnect()
