        VStack(spacing: 20) {
            Text("今天吃什麼？")

                .resizable()
                .scaledToFit()
                .frame(height: 300)
                .clipShape(RoundedRectangle(cornerRadius: 20))
                .shadow(radius: 10)
                .padding()
            
            Text(selectedFood.name)
                .font(.title)
                .fontWeight(.semibold)
            
            Text(selectedFood.description)
                .font(.body)
                .multilineTextAlignment(.center)
                .padding(.horizontal)
                .foregroundStyle(.secondary)
            
            Spacer()
            
            Button(action: {
                withAnimation {
                    selectedFood = Food.allFoods.randomElement()!

                    .foregroundStyle(.white)

最後我們再加強一下 App 的畫面，請 AI 將畫面變得超級漂亮 !

import SwiftUI

struct ContentView: View {
    @State private var selectedFood: Food = Food.allFoods.randomElement()!
    @State private var isAnimating = false
    
    // Gradient colors for background
    let backgroundColors: [Color] = [
        Color(red: 0.95, green: 0.90, blue: 0.85), // Warm beige
        Color(red: 0.90, green: 0.95, blue: 0.90), // Soft mint
        Color(red: 0.90, green: 0.90, blue: 0.95)  // Light periwinkle
    ]
    
    var body: some View {
        ZStack {
            // Animated Background
            LinearGradient(gradient: Gradient(colors: backgroundColors), startPoint: .topLeading, endPoint: .bottomTrailing)
                .hueRotation(.degrees(isAnimating ? 45 : 0))
                .ignoresSafeArea()
                .onAppear {
                    withAnimation(.easeInOut(duration: 5.0).repeatForever(autoreverses: true)) {
                        isAnimating = true
                    }
                }
            
            VStack(spacing: 25) {
                Text("今天吃什麼？")
                    .font(.system(size: 34, weight: .heavy, design: .serif))
                    .foregroundStyle(
                        LinearGradient(
                            colors: [.primary, .secondary],
                            startPoint: .topLeading,
                            endPoint: .bottomTrailing
                        )
                    )
                    .shadow(color: .black.opacity(0.1), radius: 2, x: 0, y: 2)
                    .padding(.top, 20)
                
                Spacer()
                
                // Food Card
                VStack(spacing: 0) {
                    Image(selectedFood.imageName)
                        .resizable()
                        .scaledToFit()
                        .frame(height: 280)
                        .clipShape(RoundedRectangle(cornerRadius: 20))
                        .shadow(color: .black.opacity(0.15), radius: 15, x: 0, y: 10)
                        .padding(.top, 20)
                        .padding(.horizontal, 20)
                        .id(selectedFood.id) // Important for transition
                        .transition(.asymmetric(
                            insertion: .scale(scale: 0.8).combined(with: .opacity),
                            removal: .scale(scale: 1.1).combined(with: .opacity)
                        ))
                    
                    VStack(spacing: 10) {
                        Text(selectedFood.name)
                            .font(.system(size: 28, weight: .bold, design: .rounded))
                            .foregroundStyle(.primary)
                        
                        Text(selectedFood.description)
                            .font(.system(size: 16, weight: .regular, design: .default))
                            .multilineTextAlignment(.center)
                            .foregroundStyle(.secondary)
                            .lineSpacing(4)
                            .fixedSize(horizontal: false, vertical: true)
                    }
                    .padding(25)
                }
                .background(
                    RoundedRectangle(cornerRadius: 30)
                        .fill(.ultraThinMaterial)
                        .shadow(color: .black.opacity(0.1), radius: 20, x: 0, y: 10)
                )
                .padding(.horizontal, 20)
                
                Spacer()
                
                // Action Button
                Button(action: {
                    withAnimation(.spring(response: 0.5, dampingFraction: 0.7)) {
                        selectedFood = Food.allFoods.randomElement()!
                    }
                }) {
                    HStack {
                        Image(systemName: "dice.fill")
                        Text("隨機選擇")
                    }
                    .font(.title3.bold())
                    .foregroundStyle(.white)
                    .padding(.vertical, 16)
                    .padding(.horizontal, 40)
                    .background(
                        LinearGradient(
                            colors: [Color.blue, Color.purple],
                            startPoint: .leading,
                            endPoint: .trailing
                        )
                    )
                    .clipShape(Capsule())
                    .shadow(color: .blue.opacity(0.3), radius: 10, x: 0, y: 5)
                }
                .padding(.bottom, 30)
            }
        }
    }
}
