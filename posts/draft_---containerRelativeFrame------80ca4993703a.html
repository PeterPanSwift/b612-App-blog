利用 containerRelativeFrame 平分畫面


import SwiftUI

struct ContentView: View {
    
    let zodiacSigns = [
        "牡羊座",
        "金牛座",
        "雙子座",
        "巨蟹座",
        "獅子座",
        "處女座",
        "天秤座",
        "天蠍座",
        "射手座",
        "魔羯座",
        "水瓶座",
        "雙魚座"
    ]
    
    var body: some View {
        VStack(spacing: 10) {
            ForEach(0..<4) { i in
                HStack(spacing: 10) {
                    ForEach(0..<3) { j in
                        Text(zodiacSigns[i * 3 + j])
                            .font(.largeTitle)
                            .containerRelativeFrame(.horizontal, count: 3, spacing: 10)
                            .containerRelativeFrame(.vertical, count: 4, spacing: 10)
                            .background(.yellow)
                            .clipped()
                    }
                }
            }
        }
    }
}
