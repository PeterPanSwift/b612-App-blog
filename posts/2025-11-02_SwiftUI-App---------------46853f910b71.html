SwiftUI App 實現全螢幕背景圖的幾種方法

開發 iOS App 時，有時我們會想設定全螢幕的背景圖，因此通常會從背景圖的 Image 呼叫 modifier resizable、scaledToFill、ignoresSafeArea。

不過此時卻可能遇到圖片超出螢幕， 文字被切到的問題，例如以下例子。

  • 例子 1。
import SwiftUI

struct ContentView: View {
    
    var body: some View {
        ZStack {
            Image(.dragonSaber)
                .resizable()
                .scaledToFill()
                .ignoresSafeArea()
                .opacity(0.3)
            Text("武林至尊，寶刀屠龍，號令天下，莫敢不從！ 倚天不出，誰與爭鋒？")
                .font(.largeTitle.bold())
                .padding()               
        }
    }
}
  • 例子 2。
import SwiftUI

struct ContentView: View {
   
    var body: some View {
        ZStack(alignment: .bottomTrailing) {
            Image(.prince)
                .resizable()
                .scaledToFill()
                .ignoresSafeArea()
            Text("冰島極光寫程式")
                .font(.largeTitle)
                .foregroundStyle(.white)
                .padding()
        }
    }
}

為什麼呢 ? 都是背景圖惹的禍，點選下方的箭頭切到 Selectable 模式後，可看到圖片為了維持比例填滿，它的寬度超出了螢幕，因此文字跑到了螢幕之外。

為了讓圖片的寬度等於螢幕的寬度，我們有幾種做法。

  • containerRelativeFrame(.horizontal)。
  • frame(minWidth: 0, maxWidth: .infinity)。
  • GeometryReader。

ps: 通常問題都是圖片的寬度超出螢幕，不過某些比例的圖也可能讓圖片的高度超出螢幕，此時一樣可用這三個方法修正，只是要改成調整高度。

containerRelativeFrame(.horizontal) — iOS 17 以上

containerRelativeFrame(.horizontal) 將讓圖片的寬度等於容器的寬度，對以下例子的 Image 來說，它的容器是螢幕，因此圖片的寬度將等於螢幕的寬度。

import SwiftUI

struct ContentView: View {
    
    var body: some View {
        ZStack {
            Image(.dragonSaber)
                .resizable()
                .scaledToFill()
                .containerRelativeFrame(.horizontal)
                .ignoresSafeArea()
                .opacity(0.3)
            Text("武林至尊，寶刀屠龍，號令天下，莫敢不從！ 倚天不出，誰與爭鋒？")
                .font(.largeTitle.bold())
                .padding()
               
        }
    }
}

從 Selectable 模式可看到圖片果然不再超出螢幕。

import SwiftUI

struct ContentView: View {
   
    var body: some View {
        ZStack(alignment: .bottomTrailing) {
            Image(.prince)
                .resizable()
                .scaledToFill()
                .containerRelativeFrame(.horizontal)
                .ignoresSafeArea()
            Text("冰島極光寫程式")
                .font(.largeTitle)
                .foregroundStyle(.white)
                .padding()
        }
    }
}

加上 TabView & NavigationStack 後依然是全螢幕。

struct ContentView: View {
    
    var body: some View {
        TabView {
            Tab("star", systemImage: "star") {
                NavigationStack {
                    ZStack(alignment: .bottomTrailing) {
                        Image(.prince)
                            .resizable()
                            .scaledToFill()
                            .containerRelativeFrame(.horizontal)
                            .ignoresSafeArea()
                        Text("冰島極光寫程式")
                            .font(.largeTitle)
                            .foregroundStyle(.white)
                            .padding()
                    }
                }
            }
        }
    }
}

frame(minWidth: 0, maxWidth: .infinity)

透過 frame(minWidth: 0, maxWidth: .infinity) 限制圖片的最大寬度為可用空間的寬度，因此圖片的寬度將等於螢幕的寬度。

import SwiftUI

struct ContentView: View {
    
    var body: some View {
        ZStack {
            Image(.dragonSaber)
                .resizable()
                .scaledToFill()
                .frame(minWidth: 0, maxWidth: .infinity)                
                .ignoresSafeArea()
                .opacity(0.3)
            Text("武林至尊，寶刀屠龍，號令天下，莫敢不從！ 倚天不出，誰與爭鋒？")
                .font(.largeTitle.bold())
                .padding()
        }
    }
}

GeometryReader

透過 GeometryReader 取得可用空間的寬度，在以下例子 geometryProxy.size.width 將取得螢幕的寬度。

import SwiftUI

struct ContentView: View {
    
    var body: some View {
        GeometryReader { geometryProxy in
            ZStack {
                Image(.dragonSaber)
                    .resizable()
                    .scaledToFill()
                    .frame(width: geometryProxy.size.width)
                    .ignoresSafeArea()
                    .opacity(0.3)
                Text("武林至尊，寶刀屠龍，號令天下，莫敢不從！ 倚天不出，誰與爭鋒？")
                    .font(.largeTitle.bold())
                    .padding()
            }
        }
    }
}
