struct ContentView: View {
    let colors: [Color] = [.red, .blue, .yellow, .black, .green, .pink, .orange, .purple, .gray]
   
    var body: some View {
        ScrollView(.horizontal) {
            HStack {
                ForEach(colors, id: \.self) { color in
                    color
                        .frame(width: 100, height: 100)
                        .scrollTransition(.interactive, axis: .horizontal) { content, phase in
                            content
                                .scaleEffect(1 - 0.12 * abs(phase.value))
                                .rotation3DEffect(.degrees(20 * phase.value), axis: (0, 1, 0))
                        }
                }
            }
        }
    }
}
